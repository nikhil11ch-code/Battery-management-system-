void setup() {
  Serial.begin(115200);
  while (!Serial) {
      delay(1);
  }
  uint32_t currentFrequency;
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("SSD1306 allocation failed"));
    for(;;);
  }
  if (! ina219.begin()) {
    Serial.println("Failed to find INA219 chip");
    while (1) { delay(10); }
  }
  display.display();
  delay(2000);
  display.clearDisplay();
  display.drawPixel(10, 10, SSD1306_WHITE);
  display.display();
  delay(2000);
 }
void loop() 
{
  float shuntvoltage = 0;
  float busvoltage = 0;
  float current_mA = 0;
  float loadvoltage = 0;
  float power_mW = 0;
  shuntvoltage = ina219.getShuntVoltage_mV();
  busvoltage = ina219.getBusVoltage_V();
  current_mA = ina219.getCurrent_mA();
  power_mW = ina219.getPower_mW();
  loadvoltage = busvoltage + (shuntvoltage / 1000);
  testdraw_ina219(current_mA, busvoltage);
  Serial.print(busvoltage);
  Serial.print(",");
  Serial.print(current_mA);
  Serial.println("");
  delay(2000);
}
void testdraw_ina219(float i,float v) 
{
  display.clearDisplay();
  display.setTextSize(2);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0,0);
  int vint = (int)v;
  int iint = (int)i;
  int vvv = ((int)(v*1000))%1000;
  int iii = ((int)(i*1000))%1000;
  
  display.print(F("V: ")); display.print(vint, DEC); display.print(F(".")); display.println(vvv, DEC); 
  display.print(F("I: "));
  if (i<0){
    display.print(F("-")); 
  }
  display.print(abs(iint), DEC); display.print(F(".")); display.println(abs(iii), DEC); 
  display.display();
}

#define XPOS   0
#define YPOS   1
#define DELTAY 2
void testanimate(const uint8_t *bitmap, uint8_t w, uint8_t h) 
{
  int8_t f, icons[NUMFLAKES][3];
  for(f=0; f< NUMFLAKES; f++) {
    icons[f][XPOS]   = random(1 - LOGO_WIDTH, display.width());
    icons[f][YPOS]   = -LOGO_HEIGHT;
    icons[f][DELTAY] = random(1, 6);
    Serial.print(F("x: "));
    Serial.print(icons[f][XPOS], DEC);
    Serial.print(F(" y: "));
    Serial.print(icons[f][YPOS], DEC);
    Serial.print(F(" dy: "));
    Serial.println(icons[f][DELTAY], DEC);
  }
  for(;;) 
  {
    display.clearDisplay();
    for(f=0; f< NUMFLAKES; f++) {
      display.drawBitmap(icons[f][XPOS], icons[f][YPOS], bitmap, w, h, SSD1306_WHITE);
    }
    display.display();
    delay(200);
    for(f=0; f< NUMFLAKES; f++) 
    {
      icons[f][YPOS] += icons[f][DELTAY];
      if (icons[f][YPOS] >= display.height()) {
        icons[f][XPOS]   = random(1 - LOGO_WIDTH, display.width());
        icons[f][YPOS]   = -LOGO_HEIGHT;
        icons[f][DELTAY] = random(1, 6);
      }
    }
  }
}
